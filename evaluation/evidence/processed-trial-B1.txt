--- artifact: src/cli.js ---
import { logger } from './logger.js'
import { AppError } from './errors.js'
import { config } from './config.js'

function main() {
  logger.info('startup')
  try {
    const name = process.argv[2]
    if (!name || name.trim() === '') {
      throw new AppError(1001, 'invalid input')
    }
    console.log(config.greeting_template.replace('{name}', name))
  } catch (err) {
    if (err instanceof AppError) {
      process.stderr.write(err.message + '\n')
      process.exitCode = 1
      return
    }
    throw err
  } finally {
    logger.info('shutdown')
  }
}
